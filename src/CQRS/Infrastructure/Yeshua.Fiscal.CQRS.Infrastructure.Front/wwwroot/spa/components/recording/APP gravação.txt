import 'dart:io';
import 'package:flutter/material.dart';
import 'package:flutter_sound/flutter_sound.dart';
import 'package:path_provider/path_provider.dart';
import 'package:http/http.dart' as http;
import 'package:shared_preferences/shared_preferences.dart';

void main() {
  runApp(AudioRecorderApp());
}

class AudioRecorderApp extends StatefulWidget {
  @override
  _AudioRecorderAppState createState() => _AudioRecorderAppState();
}

class _AudioRecorderAppState extends State<AudioRecorderApp> {
  FlutterSoundRecorder? _recorder;
  bool _isRecording = false;
  String? _filePath;
  
  @override
  void initState() {
    super.initState();
    _initRecorder();
  }

  Future<void> _initRecorder() async {
    _recorder = FlutterSoundRecorder();
    await _recorder!.openRecorder();
  }

  Future<void> _startRecording() async {
    Directory dir = await getApplicationDocumentsDirectory();
    _filePath = "${dir.path}/audio_${DateTime.now().millisecondsSinceEpoch}.aac";
    await _recorder!.startRecorder(toFile: _filePath, codec: Codec.aacADTS);
    setState(() => _isRecording = true);
  }

  Future<void> _stopRecording() async {
    await _recorder!.stopRecorder();
    setState(() => _isRecording = false);
    _saveAudioToQueue(_filePath!);
  }

  Future<void> _saveAudioToQueue(String path) async {
    final prefs = await SharedPreferences.getInstance();
    List<String> queue = prefs.getStringList('audioQueue') ?? [];
    queue.add(path);
    await prefs.setStringList('audioQueue', queue);
    _uploadPendingAudios();
  }

  Future<void> _uploadPendingAudios() async {
    final prefs = await SharedPreferences.getInstance();
    List<String> queue = prefs.getStringList('audioQueue') ?? [];

    for (String path in queue) {
      bool success = await _uploadAudio(path);
      if (success) {
        queue.remove(path);
        await prefs.setStringList('audioQueue', queue);
      }
    }
  }

  Future<bool> _uploadAudio(String filePath) async {
    var request = http.MultipartRequest(
      'POST', Uri.parse('https://yourserver.com/upload'),
    );
    request.files.add(await http.MultipartFile.fromPath('audio', filePath));
    var response = await request.send();
    return response.statusCode == 200;
  }

  @override
  void dispose() {
    _recorder!.closeRecorder();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(title: Text('Gravador de Áudio')),
        body: Center(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              ElevatedButton(
                onPressed: _isRecording ? _stopRecording : _startRecording,
                child: Text(_isRecording ? 'Parar Gravação' : 'Iniciar Gravação'),
              ),
              SizedBox(height: 20),
              ElevatedButton(
                onPressed: _uploadPendingAudios,
                child: Text('Forçar Upload'),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
